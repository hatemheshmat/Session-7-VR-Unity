Target Spawner Implementation

Create a TargetSpawner script to dynamically instantiate target prefabs at random positions within a defined spawn area.

using UnityEngine; using System.Collections; public class TargetSpawner : MonoBehaviour { [SerializeField] private GameObject[] targetPrefabs; [SerializeField] private Transform spawnAreaMin; [SerializeField] private Transform spawnAreaMax; [SerializeField] private float spawnInterval = 2.0f; [SerializeField] private int maxTargets = 5; private int activeTargets = 0; void Start() { StartCoroutine(SpawnTargets()); } private IEnumerator SpawnTargets() { while (true) { if (activeTargets < maxTargets) { SpawnTarget(); activeTargets++; } yield return new WaitForSeconds(spawnInterval); } } }

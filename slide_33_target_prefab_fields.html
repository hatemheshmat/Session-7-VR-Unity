Target Script Configuration

Configure each target with different point values and visual feedback when hit. Create a prefab to easily spawn multiple targets with consistent behavior.

using UnityEngine; using UnityEngine.XR.Interaction.Toolkit; public class Target : MonoBehaviour { // Configurable fields [SerializeField] private int pointValue = 10; [SerializeField] private Material defaultMaterial; [SerializeField] private Material hitMaterial; private ScoreManager scoreManager; private MeshRenderer meshRenderer; void Start() { scoreManager = FindObjectOfType<ScoreManager>(); meshRenderer = GetComponent<MeshRenderer>(); // Hook up the XR event GetComponent<XRSimpleInteractable>().selectEntered.AddListener(OnHit); } public void OnHit(SelectEnterEventArgs args) { // Add points and change material scoreManager.AddPoints(pointValue); meshRenderer.material = hitMaterial; // Reset after delay Invoke("ResetTarget", 1.0f); } }

Score Manager Script

using UnityEngine; using TMPro; public class ScoreManager : MonoBehaviour { public TextMeshProUGUI scoreText; private int currentScore = 0; public void AddScore(int points) { currentScore += points; scoreText.text = $"Score: {currentScore}"; } }

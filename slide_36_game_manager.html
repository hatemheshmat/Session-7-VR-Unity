Game Manager Implementation

Create a GameManager script to coordinate all game components and manage game state transitions. This central controller ensures proper communication between systems.

using UnityEngine; using TMPro; public class GameManager : MonoBehaviour { // Singleton instance public static GameManager Instance { get; private set; } // Game state public enum GameState { Menu, Playing, GameOver } private GameState currentState; // References [SerializeField] private GameTimer gameTimer; [SerializeField] private ScoreManager scoreManager; [SerializeField] private TargetSpawner targetSpawner; [SerializeField] private GameObject menuPanel; [SerializeField] private GameObject gameOverPanel; void Awake() { // Singleton pattern if (Instance == null) { Instance = this; } else if (Instance != this) { Destroy(gameObject); } } void Start() { // Subscribe to events gameTimer.OnGameEnd += HandleGameOver; // Set initial state SetGameState(GameState.Menu); } public void StartGame() { SetGameState(GameState.Playing); scoreManager.ResetScore(); gameTimer.StartGame(); targetSpawner.StartSpawning(); } }

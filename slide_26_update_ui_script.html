Connecting Scripts to TextMeshPro Elements

ScoreManager Script

Create a C# script to manage and update the score display on your UI canvas. This script will handle score tracking and UI updates.

Attach to GameObject
Reference TMP_Text
Add/subtract points
Update UI display
SCORE: 0
Implementation Steps:
  1. Create C# Script named "ScoreManager"
  2. Add to GameObject in your scene
  3. Drag TMP_Text to scoreText field in Inspector

C# Script Implementation

ScoreManager.cs
using UnityEngine;
using TMPro;

public class ScoreManager : MonoBehaviour
{
    // Reference to the TextMeshPro Text component
    [SerializeField] 
    private TMP_Text scoreText;
    
    // Current score value
    private int currentScore = 0;
    
    // Start is called before the first frame update
    void Start()
    {
        // Initialize the score display
        UpdateScoreDisplay();
    }
    
    // Add points to the score
    public void AddPoints(int points)
    {
        currentScore += points;
        UpdateScoreDisplay();
    }
    
    // Reset the score to zero
    public void ResetScore()
    {
        currentScore = 0;
        UpdateScoreDisplay();
    }
    
    // Update the TextMeshPro text with current score
    private void UpdateScoreDisplay()
    {
        if (scoreText != null)
        {
            scoreText.text = "SCORE: " + currentScore.ToString();
        }
    }
}

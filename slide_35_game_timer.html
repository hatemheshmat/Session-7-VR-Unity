Timer Implementation

Create a GameTimer script to add time pressure to your target practice game. This script manages the countdown and game state transitions.

using UnityEngine; using TMPro; using System; public class GameTimer : MonoBehaviour { // Timer configuration [SerializeField] private float gameDuration = 60f; [SerializeField] private TMP_Text timerText; // Game state private float timeRemaining; private bool isGameActive = false; // Events public event Action OnGameStart; public event Action OnGameEnd; void Start() { // Initialize timer timeRemaining = gameDuration; UpdateTimerDisplay(); } void Update() { if (isGameActive) { if (timeRemaining > 0) { // Decrease timer timeRemaining -= Time.deltaTime; UpdateTimerDisplay(); } else { // Time's up EndGame(); } } } public void StartGame() { timeRemaining = gameDuration; isGameActive = true; OnGameStart?.Invoke(); } }
